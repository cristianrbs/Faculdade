#include <stdio.h>
#include <stdlib.h>


/* Função pra trocar dois elementos de lugar.
 Usa ponteiros porque senão a função receberia só uma cópia dos valores
 e a troca não refletiria no vetor original.
 */
void swap(int *p1, int *p2) {
    int tmp = *p1;
    *p1 = *p2;
    *p2 = tmp;
}

void print_array(int v[], int n) {
    printf("[");
    for (int i = 0; i < n; i++) {
        printf("%d", v[i]);
        if (i < n - 1) printf(", ");
    }
    printf("]\n");
}

void insertion_sort(int s[], int n) {
    int i = 0, j = 0;
    for (i = 1; i < n; i++) {
        j = i;
        while ((j > 0) && (s[j] < s[j-1])) {
            swap(&s[j], &s[j-1]);
            j = j - 1;
        }
    }
}

void insertionSortDuplo(int vetor[], int n) {
    int i, j;

    for (i = 1; i < n; i += 2) {
        if (i + 1 >= n) {
            j = i;
            while ((j > 0) && (vetor[j] < vetor[j-1])) {
                swap(&vetor[j], &vetor[j-1]);
                j--;
            }
            break;
        }
        if (vetor[i] > vetor[i+1]) {
            swap(&vetor[i], &vetor[i+1]);
        }
        
        j = i;
        while ((j > 0) && (vetor[j] < vetor[j-1])) {
            swap(&vetor[j], &vetor[j-1]);
            j--;
        }
        
        j = i + 1;
        while ((j > 0) && (vetor[j] < vetor[j-1])) {
            swap(&vetor[j], &vetor[j-1]);
            j--;
        }
    }
}

void testar(const char *descricao, int vetor[], int n) {
    printf("--- %s ---\n", descricao);
    printf("Antes:  ");
    print_array(vetor, n);
    insertionSortDuplo(vetor, n);
    printf("Depois: ");
    print_array(vetor, n);
    printf("\n");
}

int main() {

    //Exemplo Enunciado
    int v1[] = {7, 3, 5, 1, 4, 2};
    testar("Exemplo do enunciado (n=6, par)", v1, 6);

    return 0;
}
